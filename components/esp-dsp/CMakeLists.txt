# ============================================================================
# 本仓库的本地修改（2026-11，FakePodNanoNightPlayer）：
#   ESP32-S31 是 **RISC-V** 目标，而 esp-dsp 上游把 Xtensa（AE32/AES3）汇编源
#   无条件列进 SRCS；在 riscv32-esp-elf 下汇编这些 .S 会直接失败。
#   S31 上 esp-dsp 的 Kconfig 也会把实现选成 ANSI C（DSP_OPTIMIZATIONS_SUPPORTED
#   只对 ESP32/ESP32S3 为真），所以汇编实现本来就用不到。
#   本项目只用 `dsps_dotprod_f32`（重采样器的多相点积），因此这里只编译
#   ANSI 版 dotprod 一个源文件：
#     · 避免 .S 汇编失败；
#     · 避开 kalman/ekf 等模块在新工具链（C++26，缺 <cmath>）下的编译错误；
#     · 全量编译单元从 300+ 降到 1，构建时间几乎不受影响。
#   ⚠ 升级 esp-dsp 时请保留本段；将来要用其它模块（FIR/IIR/FFT）时，
#     把它们对应的 ANSI C 源单独加进 INCLUDE 正则即可。
# ============================================================================
list(FILTER srcs INCLUDE REGEX "modules/dotprod/float/dsps_dotprod_f32_ansi\\.c$")

idf_component_register(SRCS ${srcs}
                      INCLUDE_DIRS ${include_dirs}
                      PRIV_INCLUDE_DIRS ${priv_include_dirs})
